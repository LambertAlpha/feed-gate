# Feed Gate Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 一个 Chrome MV3 扩展：打开推特、B站、知乎、小红书首页时替换为「你来干什么？」闸门页，只能搜索或去功能页，不能进推荐流。

**Architecture:** `extension/sites.js` 是唯一配置源（首页正则、搜索前缀、快捷入口）。首页正则同时生成 declarativeNetRequest 动态重定向规则（拦整页加载），并供 background 里的 webNavigation 监听使用（拦 SPA pushState 和 bfcache 后退）。闸门页是扩展内页面，用 `chrome.storage.local` 记每日计数。

**Tech Stack:** 原生 JS ES Module（无构建）、Chrome MV3、`node --test`（单元测试）、`@playwright/test` + Playwright 自带 Chromium（端到端）、pnpm。

**设计文档:** `docs/superpowers/specs/2026-10-11-feed-gate-design.md`

**提交约定:** 每次 commit 末尾带 `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`（下面的命令已包含）。

```js
// 唯一配置源。homepage 正则同时用作 DNR 的 regexFilter（RE2）和 JS 的 RegExp，
// 只能用两者共同支持的语法：不用前瞻/后顾、不用反向引用、仅 ASCII。
// 结尾的 ([?#].*)? 允许查询参数和锚点，但不允许更深的路径。
export const SITES = [
  {
    id: 'x',
    name: '推特',
    hosts: ['x.com', 'www.x.com', 'twitter.com', 'www.twitter.com'],
    homepage: String.raw`^https://(www\.)?(x|twitter)\.com/(home)?([?#].*)?$`,
    searchPrefix: 'https://x.com/search?q=',
    links: [
      { label: '通知', url: 'https://x.com/notifications' },
      { label: '私信', url: 'https://x.com/messages' },
      { label: '书签', url: 'https://x.com/i/bookmarks' },
    ],
  },
  {
    id: 'bilibili',
    name: 'B站',
    hosts: ['www.bilibili.com', 'bilibili.com'],
    homepage: String.raw`^https://(www\.)?bilibili\.com/([?#].*)?$`,
    searchPrefix: 'https://search.bilibili.com/all?keyword=',
    links: [
      { label: '消息', url: 'https://message.bilibili.com/' },
      { label: '稍后再看', url: 'https://www.bilibili.com/watchlater/#/list' },
      { label: '历史记录', url: 'https://www.bilibili.com/history' },
    ],
  },
  {
    id: 'zhihu',
    name: '知乎',
    hosts: ['www.zhihu.com', 'zhihu.com'],
    homepage: String.raw`^https://(www\.)?zhihu\.com/([?#].*)?$`,
    searchPrefix: 'https://www.zhihu.com/search?type=content&q=',
    links: [
      { label: '通知', url: 'https://www.zhihu.com/notifications' },
      { label: '私信', url: 'https://www.zhihu.com/messages' },
    ],
  },
  {
    id: 'xiaohongshu',
    name: '小红书',
    hosts: ['www.xiaohongshu.com', 'xiaohongshu.com'],
    // /explore 是发现页（推荐流），/explore/<笔记ID> 是笔记详情，后者不能拦
    homepage: String.raw`^https://(www\.)?xiaohongshu\.com/(explore/?)?([?#].*)?$`,
    searchPrefix: 'https://www.xiaohongshu.com/search_result?keyword=',
    links: [
      { label: '通知', url: 'https://www.xiaohongshu.com/notification' },
    ],
  },
];

export const SITES_BY_ID = Object.fromEntries(SITES.map((site) => [site.id, site]));

const HOMEPAGE_MATCHERS = SITES.map((site) => [site, new RegExp(site.homepage)]);

export function matchHomepage(url) {
  for (const [site, re] of HOMEPAGE_MATCHERS) {
    if (re.test(url)) return site;
  }
  return null;
}

export function gatePath(siteId) {
  return `/gate.html?site=${siteId}`;
}

export function searchUrlFor(site, query) {
  const trimmed = query.trim();
  return trimmed ? site.searchPrefix + encodeURIComponent(trimmed) : null;
}
```

```js
import { SITES, gatePath } from './sites.js';

export function buildRedirectRules() {
  return SITES.map((site, i) => ({
    id: i + 1,
    priority: 1,
    action: { type: 'redirect', redirect: { extensionPath: gatePath(site.id) } },
    condition: {
      regexFilter: site.homepage,
      // DNR 默认不区分大小写，JS RegExp 区分；统一成区分，两边判断才一致
      isUrlFilterCaseSensitive: true,
      resourceTypes: ['main_frame'],
    },
  }));
}
```

```js
// stats 形如 { date: 'YYYY-MM-DD', counts: { [siteId]: number } }，只保留当天
export function todayKey(date) {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, '0');
  const d = String(date.getDate()).padStart(2, '0');
  return `${y}-${m}-${d}`;
}

export function bumpStats(stats, siteId, today) {
  const counts = stats?.date === today ? { ...stats.counts } : {};
  counts[siteId] = (counts[siteId] ?? 0) + 1;
  return { date: today, counts };
}

export function totalCount(stats) {
  return Object.values(stats.counts).reduce((sum, n) => sum + n, 0);
}
```

```js
function covers(pattern, host) {
  const m = /^https:\/\/\*\.([^/]+)\/\*$/.exec(pattern);
```

```js
import { SITES, matchHomepage, gatePath } from './sites.js';
import { buildRedirectRules } from './rules.js';

// 动态规则会跨重启保留；每次全量重建，保证和 sites.js 一致
async function installRules() {
  const existing = await chrome.declarativeNetRequest.getDynamicRules();
  await chrome.declarativeNetRequest.updateDynamicRules({
    removeRuleIds: existing.map((rule) => rule.id),
    addRules: buildRedirectRules(),
  });
}

chrome.runtime.onInstalled.addListener(installRules);
chrome.runtime.onStartup.addListener(installRules);

// DNR 只拦得到真正发出的请求。SPA 站内跳转（pushState）和后退恢复缓存页（bfcache）
// 不发请求，靠这里兜底
function gateIfHomepage({ tabId, frameId, url }) {
  if (frameId !== 0) return;
  const site = matchHomepage(url);
  if (site) chrome.tabs.update(tabId, { url: chrome.runtime.getURL(gatePath(site.id)) });
}

const filter = { url: SITES.flatMap((site) => site.hosts.map((host) => ({ hostEquals: host }))) };
chrome.webNavigation.onHistoryStateUpdated.addListener(gateIfHomepage, filter);
chrome.webNavigation.onCommitted.addListener(gateIfHomepage, filter);
```

```js
import { SITES_BY_ID, searchUrlFor } from './sites.js';
import { bumpStats, todayKey, totalCount } from './stats.js';

const titleEl = document.getElementById('title');
const formEl = document.getElementById('search');
const inputEl = document.getElementById('q');
const linksEl = document.getElementById('links');
const statsEl = document.getElementById('stats');

const site = SITES_BY_ID[new URLSearchParams(location.search).get('site')] ?? null;

if (site) {
  showGate(site);
  // 统计失败不影响拦截，只是不显示次数
  recordVisit(site).catch(() => {});
} else {
  titleEl.textContent = '你来这里干什么？';
  formEl.hidden = true;
}

function showGate(site) {
  const question = `你来${site.name}干什么？`;
  document.title = question;
  titleEl.textContent = question;
  inputEl.placeholder = `搜索${site.name}`;
  for (const { label, url } of site.links) {
    const a = document.createElement('a');
    a.href = url;
    a.textContent = label;
    linksEl.append(a);
  }
  formEl.addEventListener('submit', (event) => {
    event.preventDefault();
    const url = searchUrlFor(site, inputEl.value);
    // replace：按后退不会回到闸门
    if (url) location.replace(url);
  });
  inputEl.focus();
}

async function recordVisit(site) {
  const { stats } = await chrome.storage.local.get('stats');
  const next = bumpStats(stats, site.id, todayKey(new Date()));
  await chrome.storage.local.set({ stats: next });
  statsEl.textContent = `今天第 ${next.counts[site.id]} 次打开${site.name}首页 · 四站合计 ${totalCount(next)} 次`;
}
```
